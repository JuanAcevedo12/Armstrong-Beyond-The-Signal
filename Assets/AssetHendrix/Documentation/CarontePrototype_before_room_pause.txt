 Transform FindAnchor(string key){foreach(Transform t in spacecraft.GetComponentsInChildren<Transform>(true))if(t.name==key)return t;return spacecraft;}
 void Install(string key,GameObject prefab){
 GameObject old;modules.TryGetValue(key,out old);
 if(old!=null&&old.name==prefab.name)return;
 Coroutine running;if(jobs.TryGetValue(key,out running)&&running!=null)StopCoroutine(running);
 if(old!=null)Destroy(old);
 var anchor=FindAnchor(key);var go=Instantiate(prefab,anchor);if(station!=null)foreach(var t in go.GetComponentsInChildren<Transform>(true))t.gameObject.layer=8;go.name=prefab.name;modules[key]=go;
 jobs[key]=StartCoroutine(AnimateModule(go,key));
 }
 IEnumerator AnimateModule(GameObject go,string key){
 Vector3 end=Vector3.zero,start=new Vector3(0,1.5f,-2);
 go.transform.localPosition=start;
 for(float t=0;t<1;t+=Time.deltaTime/(reduced?.01f:.7f)){
 if(!go)yield break;float v=Mathf.SmoothStep(0,1,t);go.transform.localPosition=Vector3.Lerp(start,end,v);
 go.transform.localRotation=Quaternion.Slerp(Quaternion.Euler(0,65,0),Quaternion.identity,v);yield return null;}
 if(!go)yield break;go.transform.localPosition=end;go.transform.localRotation=Quaternion.identity;
 if(key.Contains("Solar")){
 var hinge=go.transform.Find("Hinge");
 if(hinge)for(float t=0;t<1;t+=Time.deltaTime/(reduced?.01f:.65f)){hinge.localRotation=Quaternion.Euler(0,0,Mathf.Lerp(85,0,Mathf.SmoothStep(0,1,t)));yield return null;}
 if(hinge)hinge.localRotation=Quaternion.identity;}
 }
